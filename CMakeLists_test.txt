cmake_minimum_required(VERSION 3.15)
project(GeoVersionControlTests VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(mongocxx REQUIRED PATHS /usr/local/lib/cmake)
find_package(bsoncxx REQUIRED PATHS /usr/local/lib/cmake)
find_package(OpenSSL REQUIRED)
find_package(Boost REQUIRED)

include_directories(${CMAKE_SOURCE_DIR}/src)
include_directories(${Boost_INCLUDE_DIRS})

# Shared production sources (everything except the application entrypoint).
add_library(geoversion_core STATIC
    src/storage/mongodb_connection/mongodb_connection.cpp
    src/storage/bpo_storage/bpo_storage.cpp
    src/storage/cas/cas.cpp
    src/storage/situation_storage/situation_storage.cpp
    src/storage/branch_storage/branch_storage.cpp
    src/storage/version_storage/version_storage.cpp
    src/storage/delta_storage/delta_storage.cpp
    src/diff/diff_engine.cpp
    src/diff/status_engine.cpp
    src/diff/geometry_utils.cpp
    src/diff/entity_matcher.cpp
    src/diff/matchers/small_diff_matcher.cpp
    src/diff/matchers/large_diff_matcher.cpp
    src/diff/matchers/adaptive_matcher.cpp
    src/merge/merge_engine.cpp
    src/utils/logger/logger.cpp
    src/utils/uuid/uuid.cpp
)

target_link_libraries(geoversion_core
    PUBLIC
    mongo::mongocxx_shared
    mongo::bsoncxx_shared
    OpenSSL::SSL
    OpenSSL::Crypto
)

enable_testing()

# One executable + one ctest entry per test module.
set(TEST_MODULES
    test_cas
    test_situation_storage
    test_branch_storage
    test_version_storage
    test_delta_storage
    test_diff_engine
    test_status_engine
    test_entity_matcher
    test_merge_engine
)

foreach(test_module ${TEST_MODULES})
    add_executable(${test_module} tests/${test_module}.cpp)
    target_link_libraries(${test_module} PRIVATE geoversion_core)
    add_test(NAME ${test_module} COMMAND ${test_module})
endforeach()
